# RAG知识库系统 — 接口调用示例

## 一、公共说明

### 1.1 基础信息

- 开发环境地址：`http://localhost:8080`
- 统一返回格式：`{"code": 0, "message": "success", "data": ..., "timestamp": ...}`
- 认证方式：Bearer Token（Authorization请求头）
- Content-Type：`application/json`（除文件上传外）

### 1.2 常用响应码

| code | 说明 |
|------|------|
| 0 | 成功 |
| 400 | 参数错误 |
| 401 | 未认证/token过期 |
| 403 | 无权限 |
| 429 | 请求限流 |
| 500 | 系统错误 |

// JavaScript (EventSource 不适用于POST，使用 fetch + ReadableStream)
async function streamChat(question) {
  const response = await fetch('/api/qa/chat/stream', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify({ question, kbIds: [1] })
  });

  const reader = response.body.getReader();
  const decoder = new TextDecoder();

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;

    const text = decoder.decode(value);
    const lines = text.split('\n');

    for (const line of lines) {
      if (line.startsWith('data: ')) {
        const token = line.slice(6);
        appendToChat(token);  // 逐字追加到界面
      } else if (line.startsWith('event: done')) {
        // next line contains metadata JSON
      }
    }
  }
}

## 十一、完整调用流程示例

### 场景：用户上传文档 → 审核 → 问答

```bash
# 1. 登录
TOKEN=$(curl -s -X POST http://localhost:8080/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"admin123"}' \
  | grep -o '"accessToken":"[^"]*"' | cut -d'"' -f4)

AUTH="Authorization: Bearer $TOKEN"

# 2. 创建知识库
KB_ID=$(curl -s -X POST http://localhost:8080/api/knowledge-bases \
  -H "$AUTH" -H "Content-Type: application/json" \
  -d '{"kbName":"测试知识库","description":"我的第一个知识库"}' \
  | grep -o '"id":[0-9]*' | head -1 | cut -d: -f2)

echo "知识库ID: $KB_ID"

# 3. 上传文档
DOC_ID=$(curl -s -X POST http://localhost:8080/api/files/upload \
  -H "$AUTH" \
  -F "file=@test.pdf" \
  -F "kbId=$KB_ID" \
  | grep -o '"id":[0-9]*' | head -1 | cut -d: -f2)

echo "文档ID: $DOC_ID"

# 4. 等待处理完成（状态变为 PENDING_REVIEW）后审核通过
sleep 30  # 实际场景轮询状态

curl -X POST http://localhost:8080/api/review/submit \
  -H "$AUTH" -H "Content-Type: application/json" \
  -d "{\"documentId\":$DOC_ID,\"result\":\"APPROVED\",\"comment\":\"通过\"}"

# 5. 等待向量化完成后，进行问答
sleep 60

curl -X POST http://localhost:8080/api/qa/chat \
  -H "$AUTH" -H "Content-Type: application/json" \
  -d "{\"question\":\"文档的主要内容是什么?\",\"kbIds\":[$KB_ID]}"
```

### 场景：Python SDK调用示例

```python
import requests
import json

BASE_URL = "http://localhost:8080"

class RAGClient:
    def __init__(self, base_url):
        self.base_url = base_url
        self.token = None

    def login(self, username, password):
        resp = requests.post(f"{self.base_url}/api/auth/login", json={
            "username": username,
            "password": password
        })
        data = resp.json()["data"]
        self.token = data["accessToken"]
        return data

    def _headers(self):
        return {"Authorization": f"Bearer {self.token}"}

    def upload_file(self, file_path, kb_id):
        with open(file_path, "rb") as f:
            resp = requests.post(
                f"{self.base_url}/api/files/upload",
                headers=self._headers(),
                files={"file": f},
                data={"kbId": kb_id}
            )
        return resp.json()

    def chat(self, question, kb_ids, stream=False):
        url = f"{self.base_url}/api/qa/chat/stream" if stream else f"{self.base_url}/api/qa/chat"
        resp = requests.post(url, headers=self._headers(), json={
            "question": question,
            "kbIds": kb_ids
        })

        if stream:
            for line in resp.iter_lines():
                if line:
                    line = line.decode()
                    if line.startswith("data: "):
                        yield line[6:]
        else:
            return resp.json()

    def get_history(self, page=1, size=20):
        resp = requests.get(f"{self.base_url}/api/qa/history", headers=self._headers(), params={
            "page": page, "size": size
        })
        return resp.json()


# 使用示例
client = RAGClient(BASE_URL)
client.login("admin", "admin123")

# 上传文件
result = client.upload_file("document.pdf", kb_id=1)
print(f"上传成功: {result}")

# 问答
answer = client.chat("什么是RAG?", kb_ids=[1])
print(f"回答: {answer['data']['answer']}")
print(f"来源: {len(answer['data']['sourceDocs'])} 个文档片段")

# 流式问答
print("流式输出: ", end="")
for token in client.chat("解释Transformer架构", kb_ids=[1], stream=True):
    print(token, end="", flush=True)
print()
```
